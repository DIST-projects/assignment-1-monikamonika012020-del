2.Write a C program to calculate the length of a string using a pointer.

#include <stdio.h>

int main()
{
    char str[100];
    char *p;
    int length = 0;

    printf("Enter a string: ");
    scanf("%s", str);

    p = str;

    while(*p != '\0')
    {
        length++;
        p++;
    }

    printf("Length of string = %d", length);

    return 0;
}
OUTPUT:
Enter a string: HELLO
Length of string = 5

4.Create an array of structures. Use dynamic memory allocation to allocate memory space and use a structure pointer with pointer arithmetic to iterate, access, and modify

#include <stdlib.h>

struct Student
{
    int id;
    char name[20];
    float mark;
};

{
    int n, i;

    for(i = 0; i < n; i++)

7.Sort an array of floating-point numbers using the Selection Sort algorithm.
#include <stdio.h>

int main()
{
    float a[5] = {4.5, 2.1, 5.6, 1.2, 3.4};
    int i, j, min;
    float temp;

    for(i = 0; i < 4; i++)
    {
        min = i;

        for(j = i + 1; j < 5; j++)
        {
            if(a[j] < a[min])
                min = j;
        }

        temp = a[i];
        a[i] = a[min];
        a[min] = temp;
    }

    printf("Sorted array:\n");

    for(i = 0; i < 5; i++)
        printf("%.1f ", a[i]);

    return 0;
}
OUTPUT:
Sorted array:
1.2 2.1 3.4 4.5 5.6

8.Define a structure with fields such as ID, Name, and Grade. Create an array of student structures and sort the students based on a different field.

#include <stdio.h>
#include <string.h>

struct Student
{
    int id;
    char name[20];
    float grade;
};

int main()
{
    struct Student s[3], temp;
    int i, j;

    for(i = 0; i < 3; i++)
    {
        printf("Enter ID, Name and Grade: ");
        scanf("%d %s %f",
              &s[i].id, s[i].name, &s[i].grade);
    }

    /* Sort based on grade */
    for(i = 0; i < 2; i++)
    {
        for(j = i + 1; j < 3; j++)
        {
            if(s[i].grade > s[j].grade)
            {
                temp = s[i];
                s[i] = s[j];
                s[j] = temp;
            }
        }
    }

    printf("\nStudents sorted by grade:\n");

    for(i = 0; i < 3; i++)
    {
        printf("%d %s %.2f\n",
               s[i].id, s[i].name, s[i].grade);
    }

    return 0;
}

Students sorted by grade:

9.Sort a set of negative numbers using the Radix Sort algorithm.
#include <stdio.h>

void sort(int a[], int n)
{
    int i, j, temp;

    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }
}

int main()
{
    int a[6] = {-45, -12, 23, -5, 10, -30};
    int i;

    sort(a, 6);

    printf("Sorted array:\n");

    for(i = 0; i < 6; i++)
        printf("%d ", a[i]);

    return 0;
}
OUTPUT:
Sorted array:
-45 -30 -12 -5 10 23

10.Write a C program to copy one array to another array using pointers.
#include <stdio.h>

int main()
{
    int a[5] = {10, 20, 30, 40, 50};
    int b[5];
    int *p = a;
    int *q = b;
    int i;

    for(i = 0; i < 5; i++)
    {
        *q = *p;
        p++;
        q++;
    }

    printf("Copied array:\n");

    for(i = 0; i < 5; i++)
        printf("%d ", b[i]);

    return 0;
}
OUTPUT:
Copied array:
10 20 30 40 50
