1) Write a C program to construct an expression tree for a given expression.
Example: (a−b)+(c∗d)
(a−b)+(c∗d)

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

struct Node {
    char data;
    struct Node *left, *right;
};

struct Node *stack[100];
int top = -1;

struct Node *createNode(char c) {
    struct Node *n = malloc(sizeof(struct Node));
    n->data = c;
    n->left = n->right = NULL;
    return n;
}

int main(void) {
    char postfix[] = "ab-cd*+";

    for (int i = 0; postfix[i] != '\0'; i++) {
        char c = postfix[i];
        struct Node *n = createNode(c);

        if (isalnum((unsigned char)c)) {
            stack[++top] = n;
        } else {
            n->right = stack[top--];
            n->left = stack[top--];
            stack[++top] = n;
        }
    }

    printf("Root operator: %c\n", stack[top]->data);
    printf("Expression tree constructed successfully.\n");
    return 0;
}

Output:

Root operator: +
Expression tree constructed successfully.

3) Write a C program to implement a max heap.

#include <stdio.h>

void insert(int heap[], int *n, int value) {
    int i = (*n)++;
    heap[i] = value;

    while (i > 0 && heap[(i - 1) / 2] < heap[i]) {
        int p = (i - 1) / 2;
        int temp = heap[i];
        heap[i] = heap[p];
        heap[p] = temp;
        i = p;
    }
}

int main(void) {
    int heap[100], n = 0;
    int a[] = {30, 10, 50, 20, 40};

    for (int i = 0; i < 5; i++)
        insert(heap, &n, a[i]);

    printf("Max heap: ");
    for (int i = 0; i < n; i++)
        printf("%d ", heap[i]);

    printf("\n");
    return 0;
}

Output:

Max heap: 50 40 30 10 20

#define N 5

int graph[N][N], visited[N];

int hasCycle(int v, int parent) {
    visited[v] = 1;

    for (int i = 0; i < N; i++) {
        if (graph[v][i]) {
            if (!visited[i]) {
                if (hasCycle(i, v))

5) Given an N×M grid containing land and water, count the number of islands.

#include <stdio.h>

#define R 5
#define C 5

char grid[R][C] = {
    {'L','L','W','W','W'},
    {'W','L','W','W','L'},
    {'L','W','W','L','L'},
    {'W','W','W','W','W'},
    {'L','L','W','L','L'}
};

void dfs(int r, int c) {
    if (r < 0 || r >= R || c < 0 || c >= C ||
        grid[r][c] != 'L')
        return;

    grid[r][c] = 'W';

    dfs(r + 1, c);
    dfs(r - 1, c);
    dfs(r, c + 1);
    dfs(r, c - 1);
}

int main(void) {
    int count = 0;

    for (int i = 0; i < R; i++) {
        for (int j = 0; j < C; j++) {
            if (grid[i][j] == 'L') {
                count++;
                dfs(i, j);
            }
        }
    }

    printf("Number of islands: %d\n", count);
    return 0;
}

Output:

Number of islands: 5

6) Find the minimum spanning tree using Kruskal's algorithm.

#include <stdio.h>
#include <stdlib.h>

struct Edge {
    int u, v, w;
};

int parent[100];

int find(int x) {
    if (parent[x] != x)
        parent[x] = find(parent[x]);
    return parent[x];
}

int compare(const void *a, const void *b) {
    return ((struct Edge *)a)->w -
           ((struct Edge *)b)->w;
}

int main(void) {
    int V = 4, E = 5, cost = 0, count = 0;

    struct Edge edges[] = {
        {0, 1, 10},
        {0, 2, 6},
        {0, 3, 5},
        {1, 3, 15},
        {2, 3, 4}
    };

    for (int i = 0; i < V; i++)
        parent[i] = i;

    qsort(edges, E, sizeof(edges[0]), compare);

    printf("Edges in MST:\n");

    for (int i = 0; i < E && count < V - 1; i++) {
        int u = find(edges[i].u);
        int v = find(edges[i].v);

        if (u != v) {
            parent[u] = v;
            printf("%d - %d : %d\n",
                   edges[i].u, edges[i].v, edges[i].w);
            cost += edges[i].w;
            count++;
        }
    }

    if (count != V - 1)
        printf("MST does not exist\n");
    else
        printf("Minimum cost = %d\n", cost);

    return 0;
}

Output:

Edges in MST:
2 - 3 : 4
0 - 3 : 5
0 - 1 : 10

7) Use Dijkstra's algorithm to find the shortest path from a source to all vertices.

#include <stdio.h>

#define V 5
#define INF 99999

int main(void) {
    int graph[V][V] = {
        {0, 10, 0, 0, 5},
        {10, 0, 1, 0, 2},
        {0, 1, 0, 4, 0},
        {0, 0, 4, 0, 3},
        {5, 2, 0, 3, 0}
    };

    int dist[V], visited[V] = {0};
    int src = 0;

    for (int i = 0; i < V; i++)
        dist[i] = INF;

    dist[src] = 0;

    for (int i = 0; i < V - 1; i++) {
        int u = -1, min = INF;

        for (int j = 0; j < V; j++) {
            if (!visited[j] && dist[j] < min) {
                min = dist[j];
                u = j;
            }
        }

        if (u == -1)
            break;

        visited[u] = 1;

        for (int v = 0; v < V; v++) {
            if (graph[u][v] > 0 && !visited[v] &&
                dist[u] + graph[u][v] < dist[v]) {
                dist[v] = dist[u] + graph[u][v];
            }
        }
    }

    printf("Vertex\tDistance from source\n");

    for (int i = 0; i < V; i++) {
        if (dist[i] == INF)
            printf("%d\tINF\n", i);
        else
            printf("%d\t%d\n", i, dist[i]);
    }

    return 0;
}

Output:

Vertex	Distance from source
0	0
1	7
2	8
3	8
4	5

10) Write a program to determine whether a graph is bipartite.

#include <stdio.h>

#define V 5

int graph[V][V];

int isBipartite(void) {
    int color[V], queue[V];

    for (int i = 0; i < V; i++)
        color[i] = -1;

    for (int s = 0; s < V; s++) {
        if (color[s] != -1)
            continue;

        int front = 0, rear = 0;

        queue[rear++] = s;
        color[s] = 0;

        while (front < rear) {
            int u = queue[front++];

            for (int v = 0; v < V; v++) {
                if (!graph[u][v])
                    continue;

                if (color[v] == -1) {
                    color[v] = 1 - color[u];
                    queue[rear++] = v;
                } else if (color[v] == color[u]) {
                    return 0;
                }
            }
        }
    }

    return 1;
}

int main(void) {
    graph[0][1] = graph[1][0] = 1;
    graph[1][2] = graph[2][1] = 1;
    graph[2][3] = graph[3][2] = 1;
    graph[3][4] = graph[4][3] = 1;

    if (isBipartite())
        printf("Graph is Bipartite\n");
    else
        printf("Graph is Not Bipartite\n");

    return 0;
}

Output:

Graph is Bipartite
